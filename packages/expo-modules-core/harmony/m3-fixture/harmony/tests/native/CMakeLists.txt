cmake_minimum_required(VERSION 3.16)
project(m3_author_native_check LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ENABLE_COMPILE_OPTIMIZATIONS OFF CACHE BOOL "Disable RNOH PGO for fixture check" FORCE)
if(NOT CORE_ROOT OR NOT RNOH_PACKAGE_DIR)
  message(FATAL_ERROR "Set CORE_ROOT to expo-modules-core and RNOH_PACKAGE_DIR to the extracted compatible RNOH package")
endif()
set(RNOH_CPP_DIR "${RNOH_PACKAGE_DIR}/src/main/cpp")
set(RNOH_APP_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
add_subdirectory("${RNOH_CPP_DIR}" rnoh EXCLUDE_FROM_ALL)
add_subdirectory("${CORE_ROOT}/harmony/src/main/cpp" expo-core EXCLUDE_FROM_ALL)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../cpp" author-fixture EXCLUDE_FROM_ALL)
add_library(m3_author_package_check OBJECT PackageCompileTest.cpp)
target_link_libraries(m3_author_package_check PRIVATE rnoh_m3_author_fixture)
