cmake_minimum_required(VERSION 3.16)
project(expo_font_package_compile_check LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
if(NOT IS_DIRECTORY "${RNOH_PACKAGE_DIR}/src/main/cpp/third-party")
  message(FATAL_ERROR "RNOH_PACKAGE_DIR must point to the unpacked RNOH 0.82.30 HAR package")
endif()
set(RNOH_CPP_DIR "${RNOH_PACKAGE_DIR}/src/main/cpp")
set(RNOH_APP_DIR "${CMAKE_CURRENT_BINARY_DIR}/app")
set(ENABLE_COMPILE_OPTIMIZATIONS OFF CACHE BOOL "" FORCE)
file(MAKE_DIRECTORY "${RNOH_APP_DIR}")
add_subdirectory("${RNOH_CPP_DIR}" rnoh)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../src/main/cpp" expo-font)
add_executable(expo_font_package_compile_check PackageCompileTest.cpp)
target_link_libraries(expo_font_package_compile_check PRIVATE rnoh_expo_font)
