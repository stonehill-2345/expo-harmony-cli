cmake_minimum_required(VERSION 3.16)
project(lane_b_linking)
set(CMAKE_SKIP_BUILD_RPATH TRUE)
set(OH_MODULE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../oh_modules")
set(RNOH_CPP_DIR "${OH_MODULE_DIR}/@rnoh/react-native-openharmony/src/main/cpp")
set(RNOH_APP_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(CMAKE_ASM_FLAGS "-Wno-error=unused-command-line-argument -Qunused-arguments")
if(EXISTS "${RNOH_CPP_DIR}/CMakeLists.txt")
  add_subdirectory("${RNOH_CPP_DIR}" rn)
else()
  # RNOH's Release HAR ships headers and prebuilt libraries.
  set(RNOH_CPP_DIR "${OH_MODULE_DIR}/@rnoh/react-native-openharmony/include")
  include("${RNOH_CPP_DIR}/react-native-harmony.cmake")
endif()
add_subdirectory(expo_linking linking)
add_library(rnoh_app SHARED PackageProvider.cpp "${RNOH_CPP_DIR}/RNOHAppNapiBridge.cpp")
target_link_libraries(rnoh_app PUBLIC rnoh rnoh_expo_linking)
