> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 实现 env 环境检查、doctor 只读诊断、受管文件 drift 保护（sync/prebuild 覆盖前阻断）与 env/doctor 退出码分级。

**Architecture:** 检测器注册表（`TOOL_CHECKS`/`PROJECT_CHECKS` 数组，env/doctor 只是遍历器）；drift 保护挂在 `syncHarmonyAutolinking` 写入咽喉处，基于 `.expo-harmony/managed-state.json` v2 的 A 类 `generatedFiles` 整文件 hash 基线与 B 类 `managedEntries` 托管依赖条目基线；探测用非抛出的 `probeCommand`。

**Tech Stack:** TypeScript (Node >= 18.18.0)、vitest、既有依赖（chalk/json5），**不新增任何 npm 依赖**。

**设计规范:** `docs/superpowers/specs/2026-08-27-v1.2.0-diagnostics-and-protection-design.md`

## Global Constraints

- 退出码语义：0=全部通过（含 skip）、1=存在任一 fail、2=无 fail 但有 warn；仅 env/doctor 设置，其他命令退出语义零变化。
- 检测分级：`level: 'required'` 缺失=fail（node/pm/ohpm）；`level: 'build'` 缺失=warn（deveco/hvigor/hdc）；Linux 平台 deveco=skip。
- drift 默认阻断，`--force` 跳过（sync --force 新增；prebuild --force 在既有 expo --clean 映射之上同时跳过阻断）。
- hash 基线存 `.expo-harmony/managed-state.json`（gitignored，复用既有设施）；无记录 + 文件存在 = 不阻断、建立基线并 info 提示。
- 受管文件清单从 `runAutolinking().files` 派生（排除 `oh-package.json5`），不手维护第二份。
- 所有用户可见文案为中文，风格对齐 `src/utils/log.ts`。
- 提交信息用中文 conventional commits（如 `feat(env): ...`）。**按主公规则：每个 Commit 步骤执行前需主公确认（或主公开工时一次性授权批量提交）。**
- 测试命令：`pnpm vitest run <file>` 单文件、`pnpm test` 全量、`pnpm type-check`。

export interface ProbeResult {
  ok: boolean;
  stdout: string;
  stderr: string;
}

/** 非抛出探测：命令缺失、非零退出、超时一律返回 ok:false，供 env 检测消费。 */
export function probeCommand(
  file: string,
  args: string[],
  opts: { timeoutMs?: number } = {},
): ProbeResult {
  const invocation = resolveCommandInvocation(file);
  try {
    const result = spawnSync(invocation.command, args, {
      shell: invocation.shell,
      encoding: 'utf8',
      timeout: opts.timeoutMs ?? 5000,
    });
    if (result.error) {
      return { ok: false, stdout: result.stdout ?? '', stderr: result.error.message };
    }
    return { ok: result.status === 0, stdout: result.stdout ?? '', stderr: result.stderr ?? '' };
  } catch (err) {
    return { ok: false, stdout: '', stderr: err instanceof Error ? err.message : String(err) };
  }
}
```

// src/env-checks/tool-checks.ts
import * as fs from 'fs';
import * as path from 'path';
import { resolvePm } from '../lib/pkg-manager';
import type { Check, CheckContext, CheckResult } from './types';

const NODE_MIN = '18.18.0'; // 对齐 package.json engines.node

/** 轻量 semver 下界比较（不引第三方依赖），actual >= min 返回 true。 */
export function compareAtLeast(actual: string, min: string): boolean {
  const parse = (v: string) => v.replace(/^[^\d]*/, '').split('.').map(n => parseInt(n, 10) || 0);
  const [a1 = 0, a2 = 0, a3 = 0] = parse(actual);
  const [m1 = 0, m2 = 0, m3 = 0] = parse(min);
  if (a1 !== m1) return a1 > m1;
  if (a2 !== m2) return a2 > m2;
  return a3 >= m3;
}

export function nodeCheck(ctx: CheckContext): CheckResult {
  const probe = ctx.probe('node', ['--version']);
  if (!probe.ok) {
    return {
      id: 'node', label: 'Node.js', status: 'fail', level: 'required',
      detail: '未找到', hint: `安装 Node.js >= ${NODE_MIN}（https://nodejs.org）`,
    };
  }
  const version = probe.stdout.trim().replace(/^v/, '');
  if (!compareAtLeast(version, NODE_MIN)) {
    return {
      id: 'node', label: 'Node.js', status: 'fail', level: 'required',
      detail: `${version}（要求 >= ${NODE_MIN}）`, hint: '升级 Node.js 后重试',
    };
  }
  return { id: 'node', label: 'Node.js', status: 'ok', level: 'required', detail: `v${version}（要求 >= ${NODE_MIN}）` };
}

export function pmCheck(ctx: CheckContext): CheckResult {
  if (!ctx.projectRoot) {
    return { id: 'pm', label: '包管理器', status: 'skip', level: 'required', detail: '非项目目录，跳过' };
  }
  // 与 install/uninstall/codegen 共用同一 PM 判定规则（flag > lockfile > 默认 pnpm），
  // env 报告的就是 CLI 实际会用的包管理器，禁止自建第二套判定。
  const pm = resolvePm([], ctx.projectRoot);
  const probe = ctx.probe(pm, ['--version']);
  if (!probe.ok) {
    return {
      id: 'pm', label: '包管理器', status: 'fail', level: 'required',
      detail: `${pm} 未找到`, hint: `安装 ${pm} 并确认其在 PATH 中`,
    };
  }
  return { id: 'pm', label: '包管理器', status: 'ok', level: 'required', detail: `${pm} ${probe.stdout.trim()}` };
}
```

// tool-checks.ts 追加：
const OHPM_HINT = process.platform === 'win32'
  ? '$env:Path += ";C:\\Program Files\\Huawei\\DevEco Studio\\tools\\ohpm\\bin"'
  : 'export PATH="/Applications/DevEco-Studio.app/Contents/tools/ohpm/bin:$PATH"';

export function ohpmCheck(ctx: CheckContext): CheckResult {
  const probe = ctx.probe('ohpm', ['--version']);
  if (!probe.ok) {
    return { id: 'ohpm', label: 'ohpm', status: 'fail', level: 'required', detail: '未找到', hint: OHPM_HINT };
  }
  return { id: 'ohpm', label: 'ohpm', status: 'ok', level: 'required', detail: probe.stdout.trim() };
}

export function hvigorCheck(ctx: CheckContext): CheckResult {
  // 项目内 hvigorw（DevEco 标准 wrapper）存在即视为可用，优先于 PATH 探测
  if (ctx.projectRoot && ctx.existsSync(path.join(ctx.projectRoot, 'harmony', 'hvigorw'))) {
    return { id: 'hvigor', label: 'hvigor', status: 'ok', level: 'build', detail: 'harmony/hvigorw' };
  }
  const probe = ctx.probe('hvigor', ['--version']);
  if (!probe.ok) {
    return { id: 'hvigor', label: 'hvigor', status: 'warn', level: 'build', detail: '未找到', hint: 'hvigor 随 DevEco Studio 提供，HAP 构建阶段需要' };
  }
  return { id: 'hvigor', label: 'hvigor', status: 'ok', level: 'build', detail: probe.stdout.trim() };
}

export function hdcCheck(ctx: CheckContext): CheckResult {
  const file = process.env.HDC_PATH ?? 'hdc';
  const probe = ctx.probe(file, ['version']);
  if (!probe.ok) {
    return { id: 'hdc', label: 'hdc', status: 'warn', level: 'build', detail: '未找到', hint: 'hdc 位于 DevEco SDK toolchains；可设置 HDC_PATH 指向 hdc 可执行文件' };
  }
  const line = probe.stdout.split(/\r?\n/).find(l => l.includes('Ver')) ?? probe.stdout;
  return { id: 'hdc', label: 'hdc', status: 'ok', level: 'build', detail: line.trim() };
}

const DEVECO_PATHS: Partial<Record<NodeJS.Platform, string[]>> = {
  darwin: ['/Applications/DevEco-Studio.app'],
  win32: ['C:\\Program Files\\Huawei\\DevEco Studio'],
};

export function devecoCheck(ctx: CheckContext): CheckResult {
  // DEVECO_HOME 逃生口：自定义安装路径优先于固定目录（未检测到 ≠ 未安装）
  const home = process.env.DEVECO_HOME;
  if (home && ctx.existsSync(home)) {
    return { id: 'deveco', label: 'DevEco Studio', status: 'ok', level: 'build', detail: home };
  }
  const candidates = DEVECO_PATHS[process.platform];
  if (!candidates) {
    return { id: 'deveco', label: 'DevEco Studio', status: 'skip', level: 'build', detail: `${process.platform} 平台跳过` };
  }
  const found = candidates.find(p => ctx.existsSync(p));
  if (!found) {
    return { id: 'deveco', label: 'DevEco Studio', status: 'warn', level: 'build', detail: '未检测到安装目录（未检测到 ≠ 未安装）', hint: '自定义安装路径请设置 DEVECO_HOME 环境变量；默认路径安装 DevEco Studio 5.0+（https://developer.huawei.com/consumer/cn/deveco-studio/）' };
  }
  return { id: 'deveco', label: 'DevEco Studio', status: 'ok', level: 'build', detail: found };
}

// src/commands/env.ts
import * as fs from 'fs';
import * as path from 'path';
import { TOOL_CHECKS } from '../env-checks/registry';
import type { CheckContext, CheckResult } from '../env-checks/types';
import { probeCommand } from '../utils/exec';
import type { Probe } from '../env-checks/types';
import { log } from '../utils/log';

export function summarize(results: CheckResult[]): { fail: number; warn: number } {
  return {
    fail: results.filter(r => r.status === 'fail').length,
    warn: results.filter(r => r.status === 'warn').length,
  };
}

export function exitCodeFor(results: CheckResult[]): number {
  const { fail, warn } = summarize(results);
  if (fail) return 1;
  return warn ? 2 : 0;
}

function printResult(r: CheckResult): void {
  const label = r.label.padEnd(12);
  const line = `${label}${r.detail ?? ''}`;
  if (r.status === 'ok') log.success(line);
  else if (r.status === 'warn') log.warn(line);
  else if (r.status === 'fail') log.error(line);
  else log.info(line);
  if (r.hint) console.log(`    └ ${r.hint}`);
}

export async function env(_args: string[], deps: { probe?: Probe; existsSync?: (p: string) => boolean } = {}): Promise<void> {
  process.exitCode = 0; // 入口重置（规范第 5 节）：process.exitCode 是进程级全局，防同进程程序化多次调用时上次的 1/2 被本次或后续命令继承
  const cwd = process.cwd();
  const projectRoot = fs.existsSync(path.join(cwd, 'package.json')) ? cwd : null;
  const ctx: CheckContext = { projectRoot, probe: deps.probe ?? probeCommand, existsSync: deps.existsSync ?? fs.existsSync };

  log.step('环境检查');
  console.log();
  const results = TOOL_CHECKS.map(check => check(ctx));
  for (const r of results) printResult(r);

  const { fail, warn } = summarize(results);
  console.log();
  log.info(`${fail} 项失败，${warn} 项警告。退出码 ${exitCodeFor(results)}。`);
  process.exitCode = exitCodeFor(results);
}

// 类型区替换 ManagedState，新增两个基线类型：
export type GeneratedFileBaseline = {
  contentHash: string;
  cliVersion: string;
};

/** B 类基线：托管条目上次 CLI 写入值。键格式 `${path}:${dependency}`。 */
export type ManagedEntryBaseline = {
  /** oh-package.json5 相对项目根路径。 */
  path: string;
  /** 托管依赖名（npm 包全名）。 */
  dependency: string;
  /** 上次 CLI 写入的 spec（file:... 形式）。 */
  spec: string;
  cliVersion: string;
};

export type ManagedState = {
  version: 2;
  packages: Record<string, ManagedPackage>;
  /** A 类基线（autolinking 整文件写入）。 */
  generatedFiles?: Record<string, GeneratedFileBaseline>;
  /** B 类基线（oh-package 托管条目写入值）。 */
  managedEntries?: Record<string, ManagedEntryBaseline>;
};

// readManagedState 中 version 校验分支替换为：
export function readManagedState(projectRoot: string): ManagedState {
  const filePath = statePath(projectRoot);
  if (!fs.existsSync(filePath)) return { version: 2, packages: {} };
  try {
    const state = JSON.parse(fs.readFileSync(filePath, 'utf8'));
    if ((state?.version === 1 || state?.version === 2) && state.packages && typeof state.packages === 'object') {
      const normalized = state.version === 1 ? { ...state, version: 2 } : state;
      // generatedFiles 形状 + 键路径安全校验：任一非法即丢弃整个字段
      // （等效无基线，sync 放行重建），避免"比对恒不等 -> 永久 drift 阻断且
      // 用户无解"，也避免键指向项目根之外被后续读写访问。
      if (!isGeneratedFilesShapeValid(projectRoot, normalized.generatedFiles)) {
        delete normalized.generatedFiles;
      }

import * as fs from 'fs';

/**
 * 唯一"目标已存在"的 rename（state tmp → state）统一入口（规范第 6 节）。
 * Windows 的 rename 替换依赖 MOVEFILE_REPLACE_EXISTING，目标被占用（杀毒
 * 扫描、编辑器打开）时抛 EPERM/EACCES/EBUSY——瞬时占用重试通常可解。
 * 其余平台与错误直接抛（交给调用方的回滚路径）。
 */
const RETRYABLE = new Set(['EPERM', 'EACCES', 'EBUSY']);

export function renameAtomic(from: string, to: string): void {
  const maxAttempts = process.platform === 'win32' ? 3 : 1;
  let lastErr: unknown;
  for (let attempt = 1; attempt <= maxAttempts; attempt += 1) {
    try {
      fs.renameSync(from, to); // 同目录 rename，原子替换
      return;
    } catch (err) {
      lastErr = err;
      const code = (err as NodeJS.ErrnoException).code;
      if (process.platform !== 'win32' || !code || !RETRYABLE.has(code)) throw err;
    }
  }
  throw lastErr;
}

export function findGeneratedFileDrift(projectRoot: string, relPaths: string[]): string[] {

/**
 * 仅刷新 HarmonyOS 原生依赖的 autolinking 托管文件。
 * 不复制模板、不删除 harmony/，因此不会影响签名、资源和其他非托管配置。
 * 写入前做 drift 检测：受管文件被手动修改且未 --force 时阻断覆盖。
 * 保持 async 形态：当前内部为同步 runAutolinking 纯计算；1.3.0 官方 link-harmony
 * 链路为异步执行，签名先行稳定。
 */
/** sync 咽喉第二参（具名导出，规范 1.1：防参数错传）。 */
export type SyncAutolinkingOptions = {
  /** 跳过 drift 阻断。 */
  force?: boolean;
};

export async function syncHarmonyAutolinking(
  projectRoot: string,
  opts: SyncAutolinkingOptions = {},
): Promise<AutolinkingResult> {
  const harmonyDir = path.join(projectRoot, 'harmony');
  if (!fs.existsSync(harmonyDir)) {
    throw new Error(
      'harmony/ 尚未生成。请先执行：pnpm dlx expo-harmony-cli prebuild --platform harmony',
    );
  }

  // hadBaseline 仅用于首建基线的 info 提示（B 类判定在 collectDrift 内按
  // managedEntries 三态+形状退化独立进行，不借用 A 类基线作前置开关）
  const hadBaseline =
    Object.keys(readManagedState(projectRoot).generatedFiles ?? {}).length > 0;

  const result = runAutolinking({
    projectRoot,
    harmonyDir,
    mapping: HARMONY_PACKAGE_MAPPING,
  });

  // 步骤 0：上次进程在事务中途被强杀的残留检查（规范 3.3 步骤 0）。
  // .cli-tmp 无独立价值自动清理；bak+目标缺失自动恢复（bak 是唯一副本）；
  // bak+目标并存无法机判取舍 → 阻断给双出口；--force 不阻断仅清理。
  const stateTmpPath = path.join(projectRoot, '.expo-harmony', 'managed-state.json.cli-tmp');
  if (fs.existsSync(stateTmpPath)) fs.rmSync(stateTmpPath, { force: true });
  for (const file of result.files) {
    const tmpPath = `${file.path}.cli-tmp`;
    if (fs.existsSync(tmpPath)) {
      fs.rmSync(tmpPath, { force: true });
      log.info(`已清理上次中断的暂存文件：${path.relative(projectRoot, tmpPath)}`);
    }
  }
  const conflictingBaks: string[] = [];
  for (const file of result.files) {
    const bakPath = `${file.path}.cli-bak`;
    if (!fs.existsSync(bakPath)) continue;
    if (!fs.existsSync(file.path)) {
      fs.renameSync(bakPath, file.path); // 目标缺失：bak 是唯一副本，恢复（无覆盖语义）
      log.info(`已恢复上次中断的备份：${path.relative(projectRoot, file.path)}`);
    } else if (opts.force) {
      fs.rmSync(bakPath, { force: true }); // 用户已明示放弃保护，仅清理
    } else {
      conflictingBaks.push(path.relative(projectRoot, bakPath));
    }
  }
  if (conflictingBaks.length > 0) {
    throw new Error([
      '检测到上次中断遗留的备份文件，且对应目标文件同时存在（无法自动判断保留哪份）：',
      ...conflictingBaks.map(b => `  - ${b}`),
      '确认磁盘上现有内容可接受 → 执行 sync --force（自动清理残留并重建基线）；',
      '需要备份中的旧内容 → 手动将 .cli-bak 重命名回原文件名后重跑 sync。',
    ].join('\n'));
  }

  // A+B 类 drift 收集（与 install/uninstall preflight、doctor 同一实现，见下方 collectDrift）
  const report = collectDrift(projectRoot, result);
  // A 类清单（排除合并写入的 oh-package），基线记录复用
  const generatedRelPaths = result.files
    .filter(f => !f.path.endsWith('oh-package.json5'))
    .map(f => path.relative(projectRoot, f.path));

  if (hasDrift(report) && !opts.force) {
    throw new DriftError(report, 'sync'); // 不落盘
  }

  // 五阶段事务写入：暂存 tmp → 备份 bak → 替换 rename → 状态原子落账 → 清 bak
  // （③④ 失败共用同一回滚实现 rollbackReplaced，见下）。
  // ① 暂存：任一 tmp 写失败 → 清理全部 tmp，原文件未动、基线不写，干净失败。
  const staged: Array<{ tmpPath: string; targetPath: string }> = [];
  try {
    for (const file of result.files) {
      fs.mkdirSync(path.dirname(file.path), { recursive: true });
      const tmpPath = `${file.path}.cli-tmp`;
      fs.writeFileSync(tmpPath, file.content);
      staged.push({ tmpPath, targetPath: file.path });
    }
  } catch (err) {
    for (const { tmpPath } of staged) fs.rmSync(tmpPath, { force: true });
    throw new Error(
      `受管文件写入失败（已回退，磁盘文件未变动）：${err instanceof Error ? err.message : err}`,
    );
  }
  // ② 备份：已存在的目标文件先 rename 为 .cli-bak（新建文件无备份）
  const backups: Array<{ bakPath: string; targetPath: string }> = [];
  for (const { targetPath } of staged) {
    if (fs.existsSync(targetPath)) {
      const bakPath = `${targetPath}.cli-bak`;
      fs.renameSync(targetPath, bakPath);
      backups.push({ bakPath, targetPath });
    }
  }
  // ③ 替换 + ④ 状态落账共用同一回滚实现（规范 3.3 步骤 5c/5d）：
  // 回滚 = rm 已替换文件 + bak rename 回原名 + 清理剩余 tmp；
  // 回滚全部成功 → 磁盘=执行前、基线未动，普通 sync 可直接重跑；
  // 回滚也失败 → 半完成态，唯一出口 sync --force。
  const replaced: string[] = [];
  const rollbackReplaced = (cause: string): Error => {
    const rollbackErrors: string[] = [];
    for (const target of replaced) fs.rmSync(target, { force: true });
    for (const { bakPath, targetPath } of backups) {
      try {
        fs.renameSync(bakPath, targetPath);
      } catch (e) {
        rollbackErrors.push(`${bakPath} -> ${targetPath}：${e instanceof Error ? e.message : e}`);
      }
    }
    for (const { tmpPath } of staged) fs.rmSync(tmpPath, { force: true });
    const leftoverBaks = backups.filter(({ bakPath }) => fs.existsSync(bakPath));
    if (rollbackErrors.length === 0 && leftoverBaks.length === 0) {
      return new Error(`${cause}（已回滚：磁盘恢复至执行前，基线未动，可直接重跑 sync）`);
    }
    return new Error([
      `${cause}且回滚不完整，需人工介入：`,
      ...rollbackErrors.map(e => `  回滚失败：${e}`),
      '磁盘处于半完成态；执行 sync --force 可强制重建（会覆盖未回滚成功的文件）。',
    ].join('\n'));
  };
  // ③ 替换：逐个 rename tmp → 目标（同目录 rename，原子且几乎不失败）
  try {
    for (const { tmpPath, targetPath } of staged) {
      fs.renameSync(tmpPath, targetPath);
      replaced.push(targetPath);
    }
  } catch (err) {
    throw rollbackReplaced(`受管文件替换失败：${err instanceof Error ? err.message : err}`);
  }
  // ④ 状态落账（原子，先于删 bak）：内存一次合成完整新 state（A 类基线 +
  // B 类 managedEntries，A/B 不分两次落盘），writeManagedState 内部 tmp+rename 落盘；
  // 失败走与替换阶段同一回滚——禁止"先删 bak 再写 state"：state 写失败会留下
  // "新文件内容 + 旧基线"的自造 drift 死锁（规范 3.3 步骤 5d）。
  let nextState = readManagedState(projectRoot);
  nextState = withGeneratedFileBaselines(
    projectRoot, nextState, generatedRelPaths, packageJson.version);
  nextState = withManagedEntries(
    nextState,
    result.managedOhPackageEntries.flatMap(({ ohPackagePath, expected }) =>
      Object.entries(expected).map(([dependency, spec]) => ({
        path: path.relative(projectRoot, ohPackagePath),
        dependency,
        spec,
      })),
    ),
    packageJson.version,
  );
  try {
    writeManagedState(projectRoot, nextState);
  } catch (err) {
    throw rollbackReplaced(`基线状态写入失败：${err instanceof Error ? err.message : err}`);
  }
  // ⑤ 清理：state 确认持久化成功后才删 bak；此步失败仅 warn（文件与基线已一致，残留无害）
  for (const { bakPath } of backups) fs.rmSync(bakPath, { force: true });
  if (!hadBaseline && generatedRelPaths.length > 0) {
    log.info('未找到受管文件基线，本次同步已建立（新 clone 或 CI 首次属正常现象）。');
  }

  return result;
}

/**
 * A+B 类 drift 收集（sync 咽喉 / install·uninstall preflight / doctor 共用同一实现）。
 * A 类：整文件 hash 比对（oh-package 是合并写入，排除在整文件基线外）。
 * B 类三态判定（现值 = 磁盘 spec；期望 = 当前映射派生值；基线 = managedEntries
 * 记录的上次 CLI 写入值）：
 *   一致 → 通过；现值=基线 → CLI 升级/映射变化的合法 spec 变化，放行覆盖；
 *   现值≠基线 → 用户手改，阻断；无基线记录 → 形状退化（file: 家族放行，
 *   脱离 file: 形式阻断）；现值 undefined（被删）→ 放行，sync 重建。
 * 与 A 类"升级不误报"哲学对齐；B 类不借用 A 类基线作前置开关。
 */
/** 单条 B 类漂移（结构化报告：调用方各自渲染文案，判定只有一份）。 */
export type DriftedEntry = {
  /** oh-package.json5 相对项目根路径。 */
  path: string;
  /** 托管依赖名。 */
  dependency: string;
  /**
   * 磁盘现值。类型故意不含 undefined：条目被删（源头值为 undefined）按
   * 规则 5 放行重建、不产生报告条目，故报告内恒有值，消费方无需判空。
   */
  currentSpec: string;
  /** 当前映射派生期望值。 */
  expectedSpec: string;
  /** managedEntries 基线值（形状退化路径无）。 */
  baselineSpec?: string;
};

export type DriftReport = { files: string[]; entries: DriftedEntry[] };

export function hasDrift(report: DriftReport): boolean {
  return report.files.length > 0 || report.entries.length > 0;
}

export function collectDrift(projectRoot: string, result: AutolinkingResult): DriftReport {
  const generatedRelPaths = result.files
    .filter(f => !f.path.endsWith('oh-package.json5'))
    .map(f => path.relative(projectRoot, f.path));
  const report: DriftReport = {
    files: findGeneratedFileDrift(projectRoot, generatedRelPaths),
    entries: [],
  };

  const entryBaselines = readManagedEntries(projectRoot);
  for (const { ohPackagePath, expected } of result.managedOhPackageEntries) {
    if (!fs.existsSync(ohPackagePath)) continue;
    let deps: Record<string, string> = {};
    try {
      deps = (JSON5.parse(fs.readFileSync(ohPackagePath, 'utf8')).dependencies ?? {}) as Record<string, string>;
    } catch {
      continue; // 解析失败交给写入流程报错
    }
    const rel = path.relative(projectRoot, ohPackagePath);
    for (const [name, want] of Object.entries(expected)) {
      const current = deps[name];
      if (current === undefined || current === want) continue;
      const baseline = entryBaselines[`${rel}:${name}`];
      if (baseline && current === baseline.spec) continue; // 合法升级，放行覆盖
      if (!baseline && /^file:.*node_modules/.test(current)) continue; // 形状退化
      report.entries.push({
        path: rel,
        dependency: name,
        currentSpec: current,
        expectedSpec: want,
        ...(baseline ? { baselineSpec: baseline.spec } : {}),
      });
    }
  }
  return report;
}

/**
 * 只读 drift preflight（install/uninstall 一切变更动作之前调用）。
 * 与 sync 咽喉同一比对实现（collectDrift，含 B 类三态与形状退化），
 * 数据源为纯计算的 runAutolinking（不落盘）。
 * runAutolinking 不可计算（oh-package 缺失等）时退化为仅 A 类基线键比对
 * （B 类跳过，保守方向为少拦不误拦——sync 咽喉会再拦），
 * preflight 绝不因自身故障反向卡死安装。
 */
export function assertNoDrift(projectRoot: string): void {
  const harmonyDir = path.join(projectRoot, 'harmony');
  if (!fs.existsSync(harmonyDir)) return; // 无 harmony/ 时本命令不会触发 sync
  let report: DriftReport;
  try {
    report = collectDrift(projectRoot, runAutolinking({
      projectRoot,
      harmonyDir,
      mapping: HARMONY_PACKAGE_MAPPING,
    }));
  } catch {
    const baselineKeys = Object.keys(readManagedState(projectRoot).generatedFiles ?? {});
    report = { files: findGeneratedFileDrift(projectRoot, baselineKeys), entries: [] };
  }
  if (hasDrift(report)) {
    throw new DriftError(report, 'preflight');
  }
}

// src/env-checks/project-checks.ts
import * as fs from 'fs';
import * as path from 'path';
import { VERSION_MATRIX } from '../version-matrix';
import { readManagedState } from '../lifecycle/managed-state';
import { runAutolinking } from '../harmony-project/autolinking';
import { collectDrift, hasDrift, type DriftReport } from '../harmony-project/standalone';
import { HARMONY_PACKAGE_MAPPING } from '../harmony-project/harmony-package-mapping';
import type { Check, CheckContext, CheckResult } from './types';

export function harmonyExistsCheck(ctx: CheckContext): CheckResult {
  if (!ctx.projectRoot) {
    return { id: 'harmony', label: 'harmony/ 工程', status: 'fail', level: 'required', detail: '非项目目录', hint: 'cd 到项目根后重试' };
  }
  if (fs.existsSync(path.join(ctx.projectRoot, 'harmony'))) {
    return { id: 'harmony', label: 'harmony/ 工程', status: 'ok', level: 'required' };
  }
  return {
    id: 'harmony', label: 'harmony/ 工程', status: 'fail', level: 'required',
    detail: 'harmony/ 不存在', hint: 'pnpm dlx expo-harmony-cli prebuild --platform harmony',
  };
}

function scopeOf(version: string, scope: 'major' | 'majorMinor'): string {
  const parts = version.replace(/^[^\d]*/, '').split('.').map(n => parseInt(n, 10) || 0);
  return scope === 'major' ? String(parts[0]) : `${parts[0]}.${parts[1]}`;
}

const BASELINE_FIELDS: Array<{ dep: string; matrix: string; scope: 'major' | 'majorMinor' }> = [
  { dep: 'react', matrix: VERSION_MATRIX.react, scope: 'majorMinor' },
  { dep: 'react-native', matrix: VERSION_MATRIX.reactNative, scope: 'majorMinor' },
  { dep: '@react-native-oh/react-native-harmony', matrix: VERSION_MATRIX.rnoh, scope: 'majorMinor' },
  { dep: 'expo', matrix: VERSION_MATRIX.expo, scope: 'major' },
];

export function baselineCheck(ctx: CheckContext): CheckResult {
  if (!ctx.projectRoot) {
    return { id: 'baseline', label: '依赖基线', status: 'skip', level: 'required', detail: '非项目目录，跳过' };
  }
  // doctor 绝不因项目文件异常而崩溃：解析失败 = warn
  let pkg: { dependencies?: Record<string, string>; devDependencies?: Record<string, string> };
  try {
    pkg = JSON.parse(fs.readFileSync(path.join(ctx.projectRoot, 'package.json'), 'utf8'));
  } catch {
    return { id: 'baseline', label: '依赖基线', status: 'warn', level: 'required', detail: '无法解析依赖清单（package.json）' };
  }
  const deps = { ...pkg.devDependencies, ...pkg.dependencies };
  const mismatched = BASELINE_FIELDS
    .filter(f => typeof deps[f.dep] === 'string' && scopeOf(deps[f.dep], f.scope) !== scopeOf(f.matrix, f.scope))
    .map(f => `${f.dep} ${scopeOf(deps[f.dep], f.scope)} ≠ 验证基线 ${scopeOf(f.matrix, f.scope)}`);
  if (mismatched.length === 0) {
    return { id: 'baseline', label: '依赖基线', status: 'ok', level: 'required', detail: '与验证基线一致' };
  }
  return {
    id: 'baseline', label: '依赖基线', status: 'warn', level: 'required',
    detail: `${mismatched.join('；')}（未验证组合）`, hint: '偏离可能引入兼容问题；如需回基线请参考 docs/guide.md',
  };
}

export function driftCheck(ctx: CheckContext): CheckResult {
  if (!ctx.projectRoot) {
    return { id: 'drift', label: '受管文件漂移', status: 'skip', level: 'required', detail: '非项目目录，跳过' };
  }
  const harmonyDir = path.join(ctx.projectRoot, 'harmony');
  if (!fs.existsSync(harmonyDir)) {
    return { id: 'drift', label: '受管文件漂移', status: 'ok', level: 'required', detail: 'harmony/ 不存在（下次 prebuild 将建立）' };
  }
  const hadBaseline =
    Object.keys(readManagedState(ctx.projectRoot).generatedFiles ?? {}).length > 0;
  if (!hadBaseline) {
    return { id: 'drift', label: '受管文件漂移', status: 'ok', level: 'required', detail: '无基线记录（下次 sync 将建立）' };
  }
  let report: DriftReport;
  try {
    // A+B 等价只读检查：与 sync 咽喉同一数据源（runAutolinking 纯计算，不落盘）
    // 与同一比对实现（collectDrift），杜绝"doctor 报无漂移、sync 却阻断"的裂缝
    report = collectDrift(ctx.projectRoot, runAutolinking({
      projectRoot: ctx.projectRoot,
      harmonyDir,
      mapping: HARMONY_PACKAGE_MAPPING,
    }));
  } catch {
    return {
      id: 'drift', label: '受管文件漂移', status: 'skip', level: 'required',
      detail: '无法计算托管条目期望（node_modules 或 oh-package 缺失）',
    };
  }
  if (!hasDrift(report)) {
    return { id: 'drift', label: '受管文件漂移', status: 'ok', level: 'required', detail: '与基线一致' };
  }
  return {
    id: 'drift', label: '受管文件漂移', status: 'warn', level: 'required',
    detail: `${report.files.length} 个文件、${report.entries.length} 处托管条目漂移，下次 sync 将被阻断`,
    hint: 'git diff 查看改动；sync --force 覆盖，或还原手动修改',
  };
}

// src/commands/doctor.ts
import * as fs from 'fs';
import * as path from 'path';
import { TOOL_CHECKS, PROJECT_CHECKS } from '../env-checks/registry';
import type { CheckContext, CheckResult } from '../env-checks/types';
import type { Probe } from '../env-checks/types';
import { probeCommand } from '../utils/exec';
import { log } from '../utils/log';
import { summarize, exitCodeFor } from './env';

function printResult(r: CheckResult): void {
  const label = r.label.padEnd(12);
  const line = `${label}${r.detail ?? ''}`;
  if (r.status === 'ok') log.success(line);
  else if (r.status === 'warn') log.warn(line);
  else if (r.status === 'fail') log.error(line);
  else log.info(line);
  if (r.hint) console.log(`    └ ${r.hint}`);
}

export async function doctor(_args: string[], deps: { probe?: Probe; existsSync?: (p: string) => boolean } = {}): Promise<void> {
  process.exitCode = 0; // 入口重置（规范第 5 节），与 env 同理
  const cwd = process.cwd();
  const projectRoot = fs.existsSync(path.join(cwd, 'package.json')) ? cwd : null;
  const ctx: CheckContext = { projectRoot, probe: deps.probe ?? probeCommand, existsSync: deps.existsSync ?? fs.existsSync };

  log.step('doctor：项目与环境诊断');
  console.log();

  const toolResults = TOOL_CHECKS.map(check => check(ctx));
  const toolSummary = summarize(toolResults);
  const okCount = toolResults.filter(r => r.status === 'ok').length;
  log.info(`环境    ✓ ${okCount} / ⚠ ${toolSummary.warn} / ✗ ${toolSummary.fail}`);

  const projectResults = PROJECT_CHECKS.map(check => check(ctx));
  for (const r of projectResults) printResult(r);

  const all = [...toolResults, ...projectResults];
  const { fail, warn } = summarize(all);
  console.log();
  if (fail > 0) log.error(`存在 ${fail} 项失败。`);
  else if (warn > 0) log.warn(`可继续开发，但有 ${warn} 项警告。`);
  else log.success('全部通过。');
  const firstIssue = all.find(r => (r.status === 'fail' || r.status === 'warn') && r.hint);
  if (firstIssue) log.info(`下一步：${firstIssue.hint}`);
  process.exitCode = exitCodeFor(all);
}
